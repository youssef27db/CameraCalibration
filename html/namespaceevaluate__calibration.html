95 rot_err = []

104 # Extrinsics
105 R_est = np.array(extr_est[cam]["rotationMatrix"], dtype=float)
106 R_gt = np.array(gt[cam]["rotationMatrix"], dtype=float)
107
108 T_est = as_vec3(extr_est[cam]["translationVector"])
109 T_gt = as_vec3(gt[cam]["translationVector"])
110

112 R_err = R_est @ R_gt.T
113 ang = rot_angle_deg(R_err)
114

124
125 # Mean values
126 mean_dT = float(np.mean(trans_err)) if trans_err else float("nan")
127 mean_dR = float(np.mean(rot_err)) if rot_err else float("nan")
128 mean_reproj = float(np.mean(reproj_err)) if reproj_err else float("nan")
129

55def rot_angle_deg(R):
56 """Extract rotation angle (degrees) from 3x3 rotation matrix."""
57 tr = np.trace(R)
58 c = (tr - 1.0) / 2.0
59 c = np.clip(c, -1.0, 1.0)
60 return float(np.degrees(np.arccos(c)))
