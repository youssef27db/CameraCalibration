56def replace_yaml_key_line(text: str, key: str, new_val: str) -> str:
57 """
58 @brief Replace YAML key-value lines.

86def run_lifcal_recalib(settings_path: str, fixed_params_path: str, store_dir: str) -> int:
87 """
88 @brief Run LiFCal recalibration and return exit code.
89
90 @param settings_path Path to Settings.yaml file
91 @param fixed_params_path Path to fixed parameters file
92 @param store_dir Directory where LiFCal will store results
93 @return Exit code from LiFCal process
94 """
95 os.makedirs(store_dir, exist_ok=True)
96
97 cmd = [LIFCAL_BIN, "recalib", settings_path, fixed_params_path]
98
99 proc = subprocess.Popen(
100 cmd,
101 stdin=subprocess.PIPE,
102 stdout=subprocess.PIPE,
103 stderr=subprocess.STDOUT,
104 text=True,
105 cwd=os.path.dirname(LIFCAL_BIN),
106 )
107
108 out, _ = proc.communicate(input=f"y\n{store_dir}\n")
109 write_text(os.path.join(store_dir, "lifcal_stdout.log"), out)
110
111 return proc.returncode

114def find_results_folder(store_dir: str) -> str:
115 """
116 @brief Find LiFCal results folder.
117
118 @param store_dir LiFCal output directory
119 @return Path to results folder (typically Calibration_Results_... subdirectory)
120 """
121 if not os.path.isdir(store_dir):
122 raise RuntimeError(f"store_dir does not exist: {store_dir}")
123
124 subdirs = []
125 for name in os.listdir(store_dir):
126 p = os.path.join(store_dir, name)
127 if os.path.isdir(p):
128 subdirs.append(p)
129
130 if not subdirs:
131 return store_dir
132
133 subdirs.sort(key=lambda p: os.path.getmtime(p))
134 return subdirs[-1]
