31 def to_serializable(self, obj):
32 """
33 @brief Convert numpy arrays and types to JSON-serializable format.
34
35 @param obj Object to convert (numpy array, dict, list, or primitive type)
36 @return JSON-serializable version of the object
37 """
38 if isinstance(obj, np.ndarray):
39 return obj.tolist()
40 if isinstance(obj, (np.floating, np.float32, np.float64)):
41 return float(obj)
42 if isinstance(obj, (np.integer, np.int32, np.int64)):
43 return int(obj)
44 if isinstance(obj, dict):
45 return {k: self.to_serializable(v) for k, v in obj.items()}
46 if isinstance(obj, (list, tuple)):
47 return [self.to_serializable(v) for v in obj]
48 return obj

50 def read_baseline_health(self, baseline_path: str) -> float:
51 """
52 @brief Extract health score from baseline.json.
53
54 @param baseline_path Path to the baseline.json file
55 @return Health score as float, or -1.0 if not found or cannot be determined
56 """
57 if not os.path.isfile(baseline_path):
58 return -1.0
59
60 try:
61 with open(baseline_path, "r", encoding="utf-8") as f:
62 base = json.load(f)
63 except Exception:
64 return -1.0
65
66 # Try direct health value from metadata
67 try:
68 h = float(base.get("meta", {}).get("health", -1.0))
69 if np.isfinite(h) and h >= 0:
70 return h
71 except Exception:
72 pass
73
74 # Fallback: compute based on calibration type
75 try:
76 from HealthMonitor import (
77 compute_initial_health_from_state,
78 compute_lifcal_health_from_combined_dict,
79 )
80 except Exception:
81 return -1.0
82
83 run_type = (base.get("meta", {}) or {}).get("runType", "")
84 state = base.get("state", {})
85
86 try:
87 if run_type == "initial":
88 rep = compute_initial_health_from_state(state)
89 return float(rep.get("health", -1.0))
90 else:
91 # Treat as lifcal/combined
92 rep = compute_lifcal_health_from_combined_dict(base)
93 return float(rep.get("health", -1.0))
94 except Exception:
95 return -1.0

98 """
99 @brief Update baseline.json with new calibration if health improved.
100
101 @param candidate_obj Dictionary containing calibration data
102 @param candidate_score Health score of the candidate calibration
103 @param method Calibration method identifier ("initial" or "lifcal")

105 baseline_path = os.path.join(self.baseDir, "baseline.json")
106 old_score = self.read_baseline_health(baseline_path)
107
108 if (not os.path.isfile(baseline_path)) or (candidate_score > old_score):
109 try:
110 with open(baseline_path, "w", encoding="utf-8") as f:
111 json.dump(candidate_obj, f, indent=2)
112 if old_score < 0:
113 self.logger.info(f"Baseline created: baseline.json (new={candidate_score:.2f}, method={method})")
114 else:
115 self.logger.info(
116 f"Baseline updated: baseline.json (old={old_score:.2f}, new={candidate_score:.2f}, method={method})"
117 )
118 except Exception as e:
119 self.logger.error(f"Failed to update baseline.json: {e}")
120 else:
121 self.logger.info(
122 f"Baseline kept: baseline.json (old={old_score:.2f} >= new={candidate_score:.2f}, method={method})"

125 def logInitialCalibration(self, calibrationState, meta=None):
126 """
127 @brief Store initial calibration result with health score.
128
129 @param calibrationState CalibrationState object with intrinsic and extrinsic parameters
130 @param meta Optional dictionary with metadata about the calibration run
131 @return Health score as float
132 """
133 if meta is None:
134 meta = {}
135
136 # Extract calibration state
137 try:
138 state_dict = calibrationState.__getState__()
139 except Exception as e:
140 self.logger.error(f"Could not extract state dict from CalibrationState: {e}")
141 return 0.0
142
143 safe_state = self.to_serializable(state_dict)
144 safe_meta = self.to_serializable(meta)
145
146 # Compute health score from metrics
147 try:
148 from HealthMonitor import compute_initial_health_from_state
149 report = compute_initial_health_from_state(state_dict)
150 score = float(report.get("health", 0.0))
151 except Exception as e:
152 self.logger.error(f"Initial HealthScore computation failed: {e}", exc_info=True)
153 score = 0.0
154
155 # Create timestamped filename
156 timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
157 imagesetnumber = meta.get("imageDir", "unknown")
158 if isinstance(imagesetnumber, str) and "imageset" in imagesetnumber:
159 imagesetnumber = imagesetnumber.split("imageset")[-1]
160
161 json_name = f"calibration_initial_imageset{imagesetnumber}_{timestamp}.json"
162 json_path = os.path.join(self.baseDir, json_name)
163
164 # Prepare output structure
165 out = {
166 "meta": safe_meta,
167 "state": safe_state
168 }
169
170 # Attach health score to metadata
171 out.setdefault("meta", {})
172 out["meta"]["health"] = score
173 out["meta"]["method"] = "initial"
174 out["meta"]["runType"] = out["meta"].get("runType", "initial")
175
176 # Save to file
177 try:
178 with open(json_path, "w", encoding="utf-8") as f:
179 json.dump(out, f, indent=2)
180 except Exception as e:
181 self.logger.error(f"Failed to write JSON result file '{json_path}': {e}")
182 return 0.0
183
184 intr = state_dict.get("intrinsics", {})
185 extr = state_dict.get("extrinsics", {})
186 self.logger.info(
187 f"Stored initial calibration to '{json_name}' "
188 f"(intrinsics for {len(intr)} cams, extrinsics for {len(extr)} cams, bundleAdjustf: {meta.get('bundleAdjust', True)})."
189 )
190 self.logger.info(f"HealthScore = {score:.2f} (method=initial)")
191
192 # Update baseline (creates baseline on first run)
193 self.update_baseline(out, score, method="initial")
194
195 return score
