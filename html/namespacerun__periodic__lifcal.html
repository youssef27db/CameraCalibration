234def run_cycle(logger):
235 """
236 @brief Execute one scheduler cycle: prepare imageset, run LiFCal, log health.
237
238 @param logger Logger instance for cycle logging
239 """
240 # 1) Only prepare imageset if needed
241 if needs_imageset_creation(logger):
242 if not os.path.isfile(RUN_IMAGESET_CREATION):
243 logger.error(f"Missing: {RUN_IMAGESET_CREATION}")
244 return
245
246 rc = run_cmd(logger, [sys.executable, RUN_IMAGESET_CREATION])
247 if rc != 0:
248 logger.error(f"run_imageset_creation failed (rc={rc}). Skipping LiFCal.")
249 return
250
251 # 2) LiFCal calibration
252 if not os.path.isfile(RUN_LIFCAL_CALIB):
253 logger.error(f"Missing: {RUN_LIFCAL_CALIB}")
254 return
255
256 rc = run_cmd(logger, [sys.executable, RUN_LIFCAL_CALIB])
257 if rc != 0:
258 logger.error(f"run_LiFCal_calibration failed (rc={rc}).")
259 return
260
261 # 3) locate combined.json (latest run)
262 latest = find_latest_run_dir(LIFCAL_OUT_ROOT)
263 cpath = combined_json_path(latest)
264
265 if not cpath:
266 logger.warning("No combined.json found after LiFCal run.")
267 return
268
269 logger.info(f"Latest combined.json: {cpath}")
270
271 # 4) log health + baseline update using your ResultLogger
272 ok = try_log_lifcal_with_resultlogger(logger, cpath)
273 if not ok:
274 logger.info("LiFCal finished, but no baseline/health logging was executed.")
