63def acquire_lock(logger) -> bool:
64 """
65 @brief Create lock file to prevent overlapping scheduler cycles.
66
67 @param logger Logger instance for status messages
68 @return True if lock was acquired, False otherwise
69 """
70 if os.path.exists(LOCK_PATH):
71 logger.warning(f"Lock exists, skipping this cycle: {LOCK_PATH}")
72 return False

74 with open(LOCK_PATH, "w") as f:

95def run_cmd(logger, cmd, cwd=None) -> int:
96 """
97 @brief Run a subprocess, log combined stdout/stderr, and return exit code.
98
99 @param logger Logger instance for command output
100 @param cmd Command as list of strings
101 @param cwd Working directory for the command (default: None)
102 @return Exit code of the subprocess
103 """
104 logger.info(f"RUN: {' '.join(cmd)}")
105 try:
106 p = subprocess.run(
107 cmd,
108 cwd=cwd,
109 stdout=subprocess.PIPE,
110 stderr=subprocess.STDOUT,
111 text=True
112 )
113 out = p.stdout or ""
114 if len(out) > 5000:
115 logger.info(out[:5000] + "\n... (truncated) ...")
116 else:
117 logger.info(out)
118 return int(p.returncode)
119 except Exception as e:
120 logger.error(f"Command failed: {e}", exc_info=True)
121 return 1

141def needs_imageset_creation(logger) -> bool:
142 """
143 @brief Check if imageset exists and has enough images per camera; trigger creation if not.
144
145 @param logger Logger instance for status messages
146 @return True if imageset needs to be created, False if it already exists
147 """
148 if not os.path.isdir(ROOT_IMAGESET):
149 logger.info("Imageset missing: ROOT_IMAGESET not found.")
150 return True
151 if not os.path.isdir(FOCUS_ROOT) or not os.path.isdir(DEPTH_ROOT):
152 logger.info("Imageset missing: focus/depth folders not found.")
153 return True
154
155 for cam in CAMERA_IDS:
156 fdir = os.path.join(FOCUS_ROOT, cam)
157 ddir = os.path.join(DEPTH_ROOT, cam)
158
159 fcnt = count_images_in_dir(fdir, exts=(".png", ".jpg", ".jpeg"))
160 dcnt = count_images_in_dir(ddir, exts=(".png", ".jpg", ".jpeg"))
161
162 if fcnt < MIN_POSES_PER_CAM:
163 logger.info(f"Imageset not ready: focus/{cam} has {fcnt} images (<{MIN_POSES_PER_CAM}).")
164 return True
165 if dcnt < MIN_POSES_PER_CAM:
166 logger.info(f"Imageset not ready: depth/{cam} has {dcnt} images (<{MIN_POSES_PER_CAM}).")
167 return True
168 logger.info("Imageset looks ready. Skipping run_imageset_creation.py.")
169 return False

172def find_latest_run_dir(out_root: str) -> str:
173 """
174 @brief Return latest run_* directory inside output root, or empty string if none.
175
176 @param out_root Root directory containing run_* folders
177 @return Path to the latest run directory, or empty string
178 """
179 if not os.path.isdir(out_root):
180 return ""
181 runs = []
182 for name in os.listdir(out_root):
183 p = os.path.join(out_root, name)
184 if os.path.isdir(p) and name.startswith("run_"):
185 runs.append(p)
186 if not runs:
187 return ""
188 runs.sort(key=lambda p: os.path.getmtime(p))
189 return runs[-1]

234def run_cycle(logger):
235 """
236 @brief Execute one scheduler cycle: prepare imageset, run LiFCal, log health.
237
238 @param logger Logger instance for cycle logging
239 """
240 # 1) Only prepare imageset if needed
241 if needs_imageset_creation(logger):
242 if not os.path.isfile(RUN_IMAGESET_CREATION):
243 logger.error(f"Missing: {RUN_IMAGESET_CREATION}")
244 return
245
246 rc = run_cmd(logger, [sys.executable, RUN_IMAGESET_CREATION])
247 if rc != 0:
248 logger.error(f"run_imageset_creation failed (rc={rc}). Skipping LiFCal.")
249 return
250
251 # 2) LiFCal calibration
252 if not os.path.isfile(RUN_LIFCAL_CALIB):
253 logger.error(f"Missing: {RUN_LIFCAL_CALIB}")
254 return
255
256 rc = run_cmd(logger, [sys.executable, RUN_LIFCAL_CALIB])
257 if rc != 0:
258 logger.error(f"run_LiFCal_calibration failed (rc={rc}).")
259 return
260
261 # 3) locate combined.json (latest run)
262 latest = find_latest_run_dir(LIFCAL_OUT_ROOT)
263 cpath = combined_json_path(latest)
264
265 if not cpath:
266 logger.warning("No combined.json found after LiFCal run.")
267 return
268
269 logger.info(f"Latest combined.json: {cpath}")
270
271 # 4) log health + baseline update using your ResultLogger
272 ok = try_log_lifcal_with_resultlogger(logger, cpath)
273 if not ok:
274 logger.info("LiFCal finished, but no baseline/health logging was executed.")

277def main():
278 """
279 @brief Periodic loop that runs LiFCal recalibration on a schedule.
280 """
281 logger = setup_logger()
282 logger.info("Starting periodic LiFCal scheduler.")
283 logger.info(f"Interval: {INTERVAL_SECONDS}s")
284 logger.info(f"Lock: {LOCK_PATH}")
285
286 while True:
287 start = time.time()
288
289 if acquire_lock(logger):
290 try:
291 logger.info("=== CYCLE START ===")
292 run_cycle(logger)
293 logger.info("=== CYCLE END ===")
294 finally:
295 release_lock(logger)
296
297 elapsed = time.time() - start
298 sleep_s = max(1.0, INTERVAL_SECONDS - elapsed)
299 logger.info(f"Sleeping {sleep_s:.1f}s\n")
300 time.sleep(sleep_s)
301
302
303if __name__ == "__main__":
304 main()
